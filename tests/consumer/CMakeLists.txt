cmake_minimum_required(VERSION 3.28)
project(velocitypp_consumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(velocitypp 0.1 CONFIG REQUIRED)
foreach(name consumer consumer_again)
  add_executable(${name} main.cpp)
  target_link_libraries(${name} PRIVATE velocitypp::velocitypp)
  if("${name}" STREQUAL "consumer_again")
    target_compile_definitions(${name} PRIVATE VELOCITYPP_CONSUMER_STOCHASTIC=1)
  endif()
  if(CONSUMER_SANITIZERS)
    target_compile_options(${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${name} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
